# Dashboard Update Cron Examples
#
# These cron job examples run the dashboard update script periodically
# to keep the dashboard data synchronized with the blockchain
#
# The update script is idempotent - it only processes new blocks and 
# skips already-processed blocks, so it's safe to run frequently
#
# Installation:
# 1. Edit your crontab: crontab -e
# 2. Add one of the examples below
# 3. Adjust the path if eIquidus is installed elsewhere


# ----------------------------------------------------------------------
# RECOMMENDED: Run every minute (ideal for Rincoin with 60s block time)
# ----------------------------------------------------------------------
* * * * * cd /opt/eiquidus && /usr/bin/node scripts/update_dashboard.js >> /var/log/eiquidus/dashboard_update.log 2>&1


# ----------------------------------------------------------------------
# Run every 5 minutes (for slower coins with 5+ minute block times)
# ----------------------------------------------------------------------
# */5 * * * * cd /opt/eiquidus && /usr/bin/node scripts/update_dashboard.js >> /var/log/eiquidus/dashboard_update.log 2>&1


# ----------------------------------------------------------------------
# Run every 10 minutes (for coins with 10+ minute block times like Bitcoin)
# ----------------------------------------------------------------------
# */10 * * * * cd /opt/eiquidus && /usr/bin/node scripts/update_dashboard.js >> /var/log/eiquidus/dashboard_update.log 2>&1


# ----------------------------------------------------------------------
# Run every hour (minimal updates, for testing or low-priority coins)
# ----------------------------------------------------------------------
# 0 * * * * cd /opt/eiquidus && /usr/bin/node scripts/update_dashboard.js >> /var/log/eiquidus/dashboard_update.log 2>&1

# ======================================================================
# HYBRID MODE: ZMQ Listener (primary) + Cron fallback
# ======================================================================
#
# When scripts/zmq_listener.js is running as a systemd service, block sync
# and dashboard updates are triggered immediately on each new block via ZMQ.
#
# In hybrid mode, keep cron jobs as a safety net but at a much lower
# frequency so they only fire if the ZMQ listener missed something:
#
#   - sync.js update     : every 15 min instead of every 1 min
#   - update_dashboard.js: every 10 min instead of every 1 min
#   - sync.js market, peers, masternodes, network-crawl, network-crawl-full: unchanged (no ZMQ equivalent)
#   - sync.js checkpoint : unchanged (long-running, scheduled job only)
#
# The existing lock files (tmp/*.pid) prevent double-execution if the
# ZMQ listener and a cron job happen to overlap.
#
# IMPORTANT: Do NOT run the ZMQ listener in BOTH eiquidus and eiquidus-test
# at the same time against the same MongoDB instance.

# Fallback: block sync every 15 minutes (ZMQ listener handles the rest)
# */15 * * * * cd /opt/eiquidus-test && /usr/bin/node scripts/sync.js update >> /var/log/eiquidus/sync_update.log 2>&1

# Fallback: dashboard update every 10 minutes
# */10 * * * * cd /opt/eiquidus-test && /usr/bin/node scripts/update_dashboard.js >> /var/log/eiquidus/dashboard_update.log 2>&1

# Market price sync every 2 minutes (cron only – external APIs, no ZMQ equivalent)
# */2 * * * * cd /opt/eiquidus-test && /usr/bin/node scripts/sync.js market >> /var/log/eiquidus/sync_market.log 2>&1

# Market history snapshot (5m bucket by default, cadence controlled by cron)
# Example: run every 5 minutes
# */5 * * * * cd /opt/eiquidus-test && /usr/bin/node scripts/sync.js market-history >> /var/log/eiquidus/sync_market_history.log 2>&1

# Peer geolocation sync every 5 minutes
# */5 * * * * cd /opt/eiquidus-test && /usr/bin/node scripts/sync.js peers >> /var/log/eiquidus/sync_peers.log 2>&1

# NOTE: when running several explorer instances on one host (e.g. production + test), give each its own
#       log file names and offset the crawler / market-history minutes between them
# Network crawler (light: verify stale/new candidates, discover further nodes via p2p addr gossip) every 30 minutes
# NOTE: separate from the "peers" job above - this independently crawls the wider p2p network rather than just the wallet's own connections
# NOTE: like the dashboard example near the top of this file, /var/log/eiquidus must exist and be writable by the cron user first
#       (sudo mkdir -p /var/log/eiquidus && sudo chown $USER:$USER /var/log/eiquidus) - otherwise the shell redirect fails silently
#       and the job never actually runs, even though cron reports no error
# */30 * * * * cd /opt/eiquidus-test && /usr/bin/node scripts/sync.js network-crawl >> /var/log/eiquidus/sync_network_crawl.log 2>&1

# Network crawler deep rescan (re-verify + re-geolocate every known node, not just new/stale ones) monthly
# 0 3 1 * * cd /opt/eiquidus-test && /usr/bin/node scripts/sync.js network-crawl-full >> /var/log/eiquidus/sync_network_crawl_full.log 2>&1

# Network crawler live-peer seed + Full-view recheck every 5 minutes (cheap: records handshake data for
# the wallet's current p2p connections, then re-handshakes - version/verack only, no address crawling -
# the handful of nodes with a successful handshake within network_crawler.aging.reachable_ttl_hours)
# NOTE: its cadence also sets how fast a dead node leaves the Full view: after
#       network_crawler.aging.unreachable_after_failures consecutive failed rechecks (2 => ~5-10 min)
# */5 * * * * cd /opt/eiquidus-test && /usr/bin/node scripts/sync.js network-crawl-seed >> /var/log/eiquidus/sync_network_crawl_seed.log 2>&1

# Masternode list refresh every 5 minutes (if masternodes_page.enabled = true)

# History checkpoint every 6 hours (unchanged – long-running batch job)
